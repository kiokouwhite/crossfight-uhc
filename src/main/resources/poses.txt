# Poses du modele articule des personnages (util/Puppet.java). Apercu : tools/pose_preview.py.
#
# Repere : X = gauche du personnage, Y = haut, Z = devant lui. Distances en pixels de skin
# (un personnage fait 32 pixels de haut), angles en degres, appliques dans l'ordre Z, X, Y.
#   rx > 0 : un bras ou une jambe part vers l'arriere, le corps ou la tete penchent vers l'avant ; rx < 0 : l'inverse
#   ry > 0 : tourne vers sa gauche ; ry < 0 : vers sa droite
#   rz > 0 : un membre s'ecarte vers la gauche ; rz < 0 : vers la droite
#
# pose <nom>
#   root   rx ry rz  [dx dy dz]     tout le corps
#   head | body | arm_r | arm_l | leg_r | leg_l   rx ry rz  [dx dy dz]
#   item   rx ry rz                 l'objet tenu dans la main droite, par rapport au bras
#                                   (0 0 0 : lame perpendiculaire au bras, comme une epee tenue)
#   item_world rx ry rz             a la place de item : orientation de l'objet par rapport au corps,
#                                   quelle que soit celle du bras (0 0 0 : lame vers l'avant, a plat sur la tranche ;
#                                   90 90 0 : lame droit vers le bas, garde de gauche a droite)
# Une partie absente reste au repos.
